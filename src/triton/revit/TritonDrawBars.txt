// ===========================================================================
//  TRITON - DRAW BARS FROM A .crm FILE - for a paste-in code runner (DevKit)
//  Revit 2021 and later. Statements only - no using lines, no namespace.
// ---------------------------------------------------------------------------
//  Asks for a Triton drawings file (.crm: Triton, Design tab > Drawings >
//  Revit), lists its views, and draws the ones you tick as detail items:
//
//    * each in its own drafting view "Triton - <section> - <view>". A view
//      that is already there is emptied and drawn again, so views placed on
//      sheets stay on them; or
//    * into the view that is open (drafting, plan, section, elevation or
//      detail), with the top left of the drawing at a point you click.
//
//  Bars and links are detail lines in the line style Triton names for their
//  size (Project tab > Drawing names), or that size's line-based detail
//  family when one is named. Cut bars are the size's detail family, else a
//  filled dot. A line style the model does not have yet is made, and the
//  summary at the end says which ones were made or not found.
// ===========================================================================

// A small JSON reader, so nothing else has to be installed: objects become
// Dictionary<string, object>, arrays List<object>, numbers double.
string text = "";
int[] at = new int[] { 0 };
System.Func<object> readValue = null;
System.Action skip = delegate ()
{ while (at[0] < text.Length && char.IsWhiteSpace(text[at[0]])) at[0]++; };
System.Func<string> readString = delegate ()
{
    var b = new System.Text.StringBuilder();
    at[0]++;   // the opening quote
    while (text[at[0]] != '"')
    {
        char c = text[at[0]++];
        if (c != '\\') { b.Append(c); continue; }
        char e = text[at[0]++];
        if (e == 'n') b.Append('\n');
        else if (e == 't') b.Append('\t');
        else if (e == 'r') b.Append('\r');
        else if (e == 'b') b.Append('\b');
        else if (e == 'f') b.Append('\f');
        else if (e == 'u')
        {
            b.Append((char)int.Parse(text.Substring(at[0], 4), System.Globalization.NumberStyles.HexNumber, inv));
            at[0] += 4;
        }
        else b.Append(e);
    }
    at[0]++;
    return b.ToString();
};
readValue = delegate ()
{
    skip();
    char c = text[at[0]];
    if (c == '{')
    {
        var d = new System.Collections.Generic.Dictionary<string, object>();
        at[0]++;
        skip();
        if (text[at[0]] == '}') { at[0]++; return d; }
        while (true)
        {
            skip();
            string key = readString();
            skip();
            at[0]++;   // the colon
            d[key] = readValue();
            skip();
            if (text[at[0]++] == '}') return d;
        }
    }
    if (c == '[')
    {
        var a = new System.Collections.Generic.List<object>();
        at[0]++;
        skip();
        if (text[at[0]] == ']') { at[0]++; return a; }
        while (true)
        {
            a.Add(readValue());
            skip();
            if (text[at[0]++] == ']') return a;
        }
    }
    if (c == '"') return readString();
    if (string.CompareOrdinal(text, at[0], "true", 0, 4) == 0) { at[0] += 4; return true; }
    if (string.CompareOrdinal(text, at[0], "false", 0, 5) == 0) { at[0] += 5; return false; }
    if (string.CompareOrdinal(text, at[0], "null", 0, 4) == 0) { at[0] += 4; return null; }
    int start = at[0];
    while (at[0] < text.Length && "+-0123456789.eE".IndexOf(text[at[0]]) >= 0) at[0]++;
    return double.Parse(text.Substring(start, at[0] - start), System.Globalization.NumberStyles.Float, inv);
};

// Reading values out of what the reader returns.
System.Func<object, string, object> get = delegate (object o, string key)
{
    var d = o as System.Collections.Generic.Dictionary<string, object>;
    object v = null;
    if (d != null) d.TryGetValue(key, out v);
    return v;
};
System.Func<object, string, string> str = delegate (object o, string key)
{
    object v = get(o, key);
    return v == null ? "" : System.Convert.ToString(v, inv);
};
System.Func<object, string, double> num = delegate (object o, string key)
{
    object v = get(o, key);
    return v is double ? (double)v : 0.0;
};
System.Func<object, string, System.Collections.Generic.List<object>> list = delegate (object o, string key)
{
    return get(o, key) as System.Collections.Generic.List<object> ?? new System.Collections.Generic.List<object>();
};
System.Func<object, string, int, double> coord = delegate (object o, string key, int i)
{
    var l = get(o, key) as System.Collections.Generic.List<object>;
    return l != null && l.Count > i && l[i] is double ? (double)l[i] : 0.0;
};

object data = null;
if (!cancelled)
{
    try
    {
        text = System.IO.File.ReadAllText(crmPath);
        data = readValue();
        if (str(data, "format") != "triton.drawings/1")
            throw new System.IO.InvalidDataException("This is not a Triton drawings file (Design tab > Drawings > Revit).");
    }
    catch (System.Exception ex)
    {
        Autodesk.Revit.UI.TaskDialog.Show("Triton", "Could not read " + crmPath + ":\n" + ex.Message);
        cancelled = true;
    }
}

        cancelled = true;
    }
}

// ------------------------------------------------------------- drawing --

if (!cancelled)
{
    var layers = get(data, "layers");
    System.Func<string, string, string> layerName = delegate (string key, string what)
    { return str(get(layers, key), what); };

    var made = new System.Collections.Generic.List<string>();
    var missing = new System.Collections.Generic.List<string>();
    var failures = new System.Collections.Generic.List<string>();
    int lines = 0, bars = 0, texts = 0, newViews = 0, redrawn = 0;
    Autodesk.Revit.DB.View firstView = null;

    // Colours for line styles this script makes, by bar size.
    var colours = new System.Collections.Generic.Dictionary<int, byte[]>();
    colours[8] = new byte[] { 128, 128, 128 }; colours[10] = new byte[] { 255, 127, 0 };
    colours[12] = new byte[] { 191, 127, 0 }; colours[16] = new byte[] { 0, 160, 0 };
    colours[20] = new byte[] { 0, 170, 170 }; colours[25] = new byte[] { 0, 0, 255 };
    colours[28] = new byte[] { 160, 0, 160 }; colours[32] = new byte[] { 220, 0, 0 };
    colours[40] = new byte[] { 120, 60, 0 };

    Autodesk.Revit.DB.Category linesCat =
        Autodesk.Revit.DB.Category.GetCategory(theDoc, Autodesk.Revit.DB.BuiltInCategory.OST_Lines);
    var subs = new System.Collections.Generic.Dictionary<string, Autodesk.Revit.DB.Category>();
    foreach (Autodesk.Revit.DB.Category c in linesCat.SubCategories) subs[c.Name] = c;
    var styles = new System.Collections.Generic.Dictionary<string, Autodesk.Revit.DB.GraphicsStyle>();
    var symbols = new System.Collections.Generic.Dictionary<string, Autodesk.Revit.DB.FamilySymbol>();

    // The line style for a layer key (bar-16, concrete, zones), made if missing.
    System.Func<string, Autodesk.Revit.DB.GraphicsStyle> style = delegate (string key)
    {
        Autodesk.Revit.DB.GraphicsStyle gs;
        if (styles.TryGetValue(key, out gs)) return gs;
        string name = layerName(key, "revit_line_style");
        if (string.IsNullOrWhiteSpace(name)) name = key;
        Autodesk.Revit.DB.Category sub;
        if (!subs.TryGetValue(name, out sub))
        {
            sub = theDoc.Settings.Categories.NewSubcategory(linesCat, name);
            int d = (int)num(get(layers, key), "diameter_mm");
            byte[] rgb = colours.ContainsKey(d) ? colours[d] : new byte[] { 0, 0, 0 };
            sub.LineColor = new Autodesk.Revit.DB.Color(rgb[0], rgb[1], rgb[2]);
            sub.SetLineWeight(key.StartsWith("bar-") ? 3 : 1, Autodesk.Revit.DB.GraphicsStyleType.Projection);
            subs[name] = sub;
            made.Add(name);
        }
        gs = sub.GetGraphicsStyle(Autodesk.Revit.DB.GraphicsStyleType.Projection);
        styles[key] = gs;
        return gs;
    };

    // A detail component type by "Family: Type", or by the type name alone.
    System.Func<string, Autodesk.Revit.DB.FamilySymbol> symbol = delegate (string name)
    {
        if (string.IsNullOrWhiteSpace(name)) return null;
        Autodesk.Revit.DB.FamilySymbol s;
        if (symbols.TryGetValue(name, out s)) return s;
        s = null;
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.FamilySymbol))
            .OfCategory(Autodesk.Revit.DB.BuiltInCategory.OST_DetailComponents))
        {
            var fs = e as Autodesk.Revit.DB.FamilySymbol;
            if (fs != null && (fs.FamilyName + ": " + fs.Name == name || fs.Name == name)) { s = fs; break; }
        }
        if (s == null) missing.Add(name);
        else if (!s.IsActive) s.Activate();
        symbols[name] = s;
        return s;
    };

    Autodesk.Revit.DB.ElementId textType = theDoc.GetDefaultElementTypeId(Autodesk.Revit.DB.ElementTypeGroup.TextNoteType);
    string textTypeName = layerName("text", "revit_text_type");
    if (!string.IsNullOrWhiteSpace(textTypeName))
    {
        bool found = false;
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.TextNoteType)))
            if (e.Name == textTypeName) { textType = e.Id; found = true; break; }
        if (!found) missing.Add(textTypeName);
    }

    Autodesk.Revit.DB.ElementId fillType = null;
    foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
        .OfClass(typeof(Autodesk.Revit.DB.FilledRegionType)))
    {
        if (fillType == null) fillType = e.Id;
        if (e.Name.ToUpperInvariant().Contains("SOLID")) { fillType = e.Id; break; }
    }

    // Where a point of a Triton view (mm) lands: origin + right * x + up * y.
    Autodesk.Revit.DB.XYZ origin = Autodesk.Revit.DB.XYZ.Zero;
    Autodesk.Revit.DB.XYZ right = Autodesk.Revit.DB.XYZ.BasisX;
    Autodesk.Revit.DB.XYZ up = Autodesk.Revit.DB.XYZ.BasisY;
    double dx = 0, dy = 0;
    System.Func<double, double, Autodesk.Revit.DB.XYZ> pt = delegate (double x, double y)
    { return origin + right.Multiply((x + dx) / FT) + up.Multiply((y + dy) / FT); };

    System.Action<Autodesk.Revit.DB.View, string, Autodesk.Revit.DB.XYZ, Autodesk.Revit.DB.XYZ> drawLine =
        delegate (Autodesk.Revit.DB.View view, string key, Autodesk.Revit.DB.XYZ p, Autodesk.Revit.DB.XYZ q)
        {
            if (p.DistanceTo(q) * FT < 1.0) return;   // Revit refuses shorter lines
            var line = Autodesk.Revit.DB.Line.CreateBound(p, q);
            var fam = symbol(layerName(key, "revit_line_type"));
            if (fam != null) theDoc.Create.NewFamilyInstance(line, fam, view);
            else theDoc.Create.NewDetailCurve(view, line).LineStyle = style(key);
            lines++;
        };

    System.Func<Autodesk.Revit.DB.XYZ, double, Autodesk.Revit.DB.Arc[]> halves =
        delegate (Autodesk.Revit.DB.XYZ c, double rMm)
        {
            return new Autodesk.Revit.DB.Arc[]
            {
                Autodesk.Revit.DB.Arc.Create(c, rMm / FT, 0, System.Math.PI, right, up),
                Autodesk.Revit.DB.Arc.Create(c, rMm / FT, System.Math.PI, 2 * System.Math.PI, right, up),
            };
        };

    System.Action<Autodesk.Revit.DB.View, string, Autodesk.Revit.DB.XYZ, double> drawCircle =
        delegate (Autodesk.Revit.DB.View view, string key, Autodesk.Revit.DB.XYZ c, double rMm)
        {
            foreach (var arc in halves(c, rMm)) theDoc.Create.NewDetailCurve(view, arc).LineStyle = style(key);
            lines++;
        };

    System.Action<Autodesk.Revit.DB.View, string, Autodesk.Revit.DB.XYZ, double> drawBar =
        delegate (Autodesk.Revit.DB.View view, string key, Autodesk.Revit.DB.XYZ c, double dMm)
        {
            bars++;
            var fam = symbol(layerName(key, "revit_section_type"));
            if (fam != null) { theDoc.Create.NewFamilyInstance(c, fam, view); return; }
            if (fillType != null)
            {
                try
                {
                    var loop = new Autodesk.Revit.DB.CurveLoop();
                    foreach (var arc in halves(c, dMm / 2)) loop.Append(arc);
                    var region = Autodesk.Revit.DB.FilledRegion.Create(theDoc, fillType, view.Id,
                        new System.Collections.Generic.List<Autodesk.Revit.DB.CurveLoop> { loop });
                    try { region.SetLineStyleId(style(key).Id); } catch { }
                    return;
                }
                catch (Autodesk.Revit.Exceptions.ApplicationException) { }
            }
            drawCircle(view, key, c, dMm / 2);
        };

    System.Action<Autodesk.Revit.DB.View, object> drawView = delegate (Autodesk.Revit.DB.View view, object v)
    {
        foreach (object it in list(v, "items"))
        {
            string type = str(it, "type"), key = str(it, "layer");
            if (type == "line")
                drawLine(view, key, pt(coord(it, "a", 0), coord(it, "a", 1)), pt(coord(it, "b", 0), coord(it, "b", 1)));
            else if (type == "rect")
            {
                double ax = coord(it, "a", 0), ay = coord(it, "a", 1), bx = coord(it, "b", 0), by = coord(it, "b", 1);
                drawLine(view, key, pt(ax, ay), pt(bx, ay));
                drawLine(view, key, pt(bx, ay), pt(bx, by));
                drawLine(view, key, pt(bx, by), pt(ax, by));
                drawLine(view, key, pt(ax, by), pt(ax, ay));
            }
            else if (type == "circle")
                drawCircle(view, key, pt(coord(it, "c", 0), coord(it, "c", 1)), num(it, "r"));
            else if (type == "bar")
                drawBar(view, key, pt(coord(it, "c", 0), coord(it, "c", 1)), num(it, "d"));
            else if (type == "text")
            {
                Autodesk.Revit.DB.TextNote.Create(theDoc, view.Id, pt(coord(it, "at", 0), coord(it, "at", 1)), str(it, "text"), textType);
                texts++;
            }
        }
    };

    // The drafting view for a Triton view: the one of the same name, emptied, or a new one.
    System.Func<object, Autodesk.Revit.DB.View> draftingView = delegate (object v)
    {
        string prefix = str(data, "view_prefix");
        if (string.IsNullOrWhiteSpace(prefix)) prefix = "Triton";
        var parts = new System.Collections.Generic.List<string> { prefix };
        if (!string.IsNullOrWhiteSpace(str(data, "section"))) parts.Add(str(data, "section"));
        parts.Add(str(v, "name"));
        string name = string.Join(" - ", parts.ToArray());
        foreach (char bad in "\\:{}[]|;<>?`~") name = name.Replace(bad, '_');
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.ViewDrafting)))
        {
            var old = e as Autodesk.Revit.DB.ViewDrafting;
            if (old == null || old.IsTemplate || old.Name != name) continue;
            var ids = new System.Collections.Generic.List<Autodesk.Revit.DB.ElementId>();
            foreach (Autodesk.Revit.DB.Element x in new Autodesk.Revit.DB.FilteredElementCollector(theDoc, old.Id)
                .WhereElementIsNotElementType())
                if (x is Autodesk.Revit.DB.CurveElement || x is Autodesk.Revit.DB.TextNote
                    || x is Autodesk.Revit.DB.FamilyInstance || x is Autodesk.Revit.DB.FilledRegion)
                    ids.Add(x.Id);
            if (ids.Count > 0) theDoc.Delete(ids);
            redrawn++;
            return old;
        }
        Autodesk.Revit.DB.ElementId typeId = null;
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.ViewFamilyType)))
        {
            var t = e as Autodesk.Revit.DB.ViewFamilyType;
            if (t != null && t.ViewFamily == Autodesk.Revit.DB.ViewFamily.Drafting) { typeId = t.Id; break; }
        }
        var view = Autodesk.Revit.DB.ViewDrafting.Create(theDoc, typeId);
        view.Name = name;
        view.Scale = System.Math.Max(1, (int)num(v, "scale"));
        newViews++;
        return view;
    };

    // Your runner may already have a transaction open; if not, this opens one.
    Autodesk.Revit.DB.Transaction tx = null;
    if (!theDoc.IsModifiable)
    {
        tx = new Autodesk.Revit.DB.Transaction(theDoc, "Triton drawings");
        tx.Start();
    }
    try
    {
        if (intoActive)
        {
            // Views side by side, rows downward, as in the AutoCAD file.
            origin = corner;
            right = active.RightDirection;
            up = active.UpDirection;
            double x = 0, y = 0, rowH = 0;
            foreach (object v in chosen)
            {
                double x0 = coord(v, "box", 0), y0 = coord(v, "box", 1), x1 = coord(v, "box", 2), y1 = coord(v, "box", 3);
                if (x > 0 && x + (x1 - x0) > 120000) { x = 0; y -= rowH + 6000; rowH = 0; }
                dx = x - x0;
                dy = y - y1;
                drawView(active, v);
                x += (x1 - x0) + 3000;
                rowH = System.Math.Max(rowH, y1 - y0);
            }
        }
        else
        {
            foreach (object v in chosen)
            {
                var view = draftingView(v);
                if (firstView == null) firstView = view;
                drawView(view, v);
            }
        }
        if (tx != null) tx.Commit();
    }
    catch (System.Exception ex)
    {
        if (tx != null && tx.HasStarted()) tx.RollBack();
        failures.Add(ex.Message);
    }

    var done = new Autodesk.Revit.UI.TaskDialog("Triton");
    if (failures.Count > 0)
    {
        done.MainInstruction = "Nothing was drawn.";
        done.MainContent = string.Join("\n", failures.ToArray());
    }
    else
    {
        done.MainInstruction = intoActive
            ? string.Format(inv, "{0} Triton view(s) drawn in {1}.", chosen.Count, active.Name)
            : string.Format(inv, "{0} drafting view(s) made, {1} drawn again.", newViews, redrawn);
        var content = new System.Text.StringBuilder();
        content.AppendLine(string.Format(inv, "{0} lines, {1} cut bars, {2} texts.", lines, bars, texts));
        if (made.Count > 0)
            content.AppendLine("\nLine styles made (not in the model yet): " + string.Join(", ", made.ToArray()));
        if (missing.Count > 0)
            content.AppendLine("\nNot found, drawn as lines or dots instead: " + string.Join(", ", missing.ToArray()));
        done.MainContent = content.ToString();
    }
    done.Show();
    if (firstView != null && failures.Count == 0)
    {
        try { theUiDoc.ActiveView = firstView; } catch { }
    }
}
