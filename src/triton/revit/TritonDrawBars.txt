// ===========================================================================
//  TRITON - DRAW BARS FROM A .crm FILE - for a paste-in code runner (DevKit)
//  Revit 2022 and later (made for 2024). Statements only - no using lines,
//  no namespace.
// ---------------------------------------------------------------------------
//  Asks for a Triton drawings file (.crm: Triton, Design tab > Drawings >
//  Revit), lists its drawings, and draws the ones you tick as detail items:
//
//    * all in ONE new drafting view for the run (the default), each drawing
//      in its frame with its caption under it and a base point (BP) to copy
//      it from, a row per element; or
//    * each in its own drafting view "Triton - <section> - <drawing>". A view
//      that is already there is emptied and drawn again, so views placed on
//      sheets stay on them; or
//    * into the view that is open, the top left at a point you click.
//
//  Bars and links are detail lines in the line style Triton names for their
//  size (T32-Reinforcement Section...). The office families are placed with
//  their parameters set: pile sections (DET_Round_Col_RFT_Dar), cut bars
//  (DET_Rebar_Dot Bar_Dar), beam links (DET_Rebar_51_Dar) and slab
//  additional bars (RFT_ADD_MODIFIED). A family that is not loaded, or
//  cannot be placed, is drawn with lines and dots instead. Dimensions are
//  Revit dimensions. The summary at the end lists what was missing and, for
//  a parameter it could not find, the parameters the family does have.
// ===========================================================================

// A small JSON reader, so nothing else has to be installed: objects become
// Dictionary<string, object>, arrays List<object>, numbers double.
string text = "";
int[] at = new int[] { 0 };
System.Func<object> readValue = null;
System.Action skip = delegate ()
{ while (at[0] < text.Length && char.IsWhiteSpace(text[at[0]])) at[0]++; };
System.Func<string> readString = delegate ()
{
    var b = new System.Text.StringBuilder();
    at[0]++;   // the opening quote
    while (text[at[0]] != '"')
    {
        char c = text[at[0]++];
        if (c != '\\') { b.Append(c); continue; }
        char e = text[at[0]++];
        if (e == 'n') b.Append('\n');
        else if (e == 't') b.Append('\t');
        else if (e == 'r') b.Append('\r');
        else if (e == 'b') b.Append('\b');
        else if (e == 'f') b.Append('\f');
        else if (e == 'u')
        {
            b.Append((char)int.Parse(text.Substring(at[0], 4), System.Globalization.NumberStyles.HexNumber, inv));
            at[0] += 4;
        }
        else b.Append(e);
    }
    at[0]++;
    return b.ToString();
};
readValue = delegate ()
{
    skip();
    char c = text[at[0]];
    if (c == '{')
    {
        var d = new System.Collections.Generic.Dictionary<string, object>();
        at[0]++;
        skip();
        if (text[at[0]] == '}') { at[0]++; return d; }
        while (true)
        {
            skip();
            string key = readString();
            skip();
            at[0]++;   // the colon
            d[key] = readValue();
            skip();
            if (text[at[0]++] == '}') return d;
        }
    }
    if (c == '[')
    {
        var a = new System.Collections.Generic.List<object>();
        at[0]++;
        skip();
        if (text[at[0]] == ']') { at[0]++; return a; }
        while (true)
        {
            a.Add(readValue());
            skip();
            if (text[at[0]++] == ']') return a;
        }
    }
    if (c == '"') return readString();
    if (string.CompareOrdinal(text, at[0], "true", 0, 4) == 0) { at[0] += 4; return true; }
    if (string.CompareOrdinal(text, at[0], "false", 0, 5) == 0) { at[0] += 5; return false; }
    if (string.CompareOrdinal(text, at[0], "null", 0, 4) == 0) { at[0] += 4; return null; }
    int start = at[0];
    while (at[0] < text.Length && "+-0123456789.eE".IndexOf(text[at[0]]) >= 0) at[0]++;
    return double.Parse(text.Substring(start, at[0] - start), System.Globalization.NumberStyles.Float, inv);
};

// Reading values out of what the reader returns.
System.Func<object, string, object> get = delegate (object o, string key)
{
    var d = o as System.Collections.Generic.Dictionary<string, object>;
    object v = null;
    if (d != null) d.TryGetValue(key, out v);
    return v;
};
System.Func<object, string, string> str = delegate (object o, string key)
{
    object v = get(o, key);
    return v == null ? "" : System.Convert.ToString(v, inv);
};
System.Func<object, string, double> num = delegate (object o, string key)
{
    object v = get(o, key);
    return v is double ? (double)v : 0.0;
};
System.Func<object, string, System.Collections.Generic.List<object>> list = delegate (object o, string key)
{
    return get(o, key) as System.Collections.Generic.List<object> ?? new System.Collections.Generic.List<object>();
};
System.Func<object, string, int, double> coord = delegate (object o, string key, int i)
{
    var l = get(o, key) as System.Collections.Generic.List<object>;
    return l != null && l.Count > i && l[i] is double ? (double)l[i] : 0.0;
};

// ------------------------------------------------------------- drawing --

if (!cancelled)
{
    var layers = get(data, "layers");
    System.Func<string, string, string> layerName = delegate (string key, string what)
    { return str(get(layers, key), what); };

    var made = new System.Collections.Generic.List<string>();
    var missing = new System.Collections.Generic.List<string>();
    var failures = new System.Collections.Generic.List<string>();
    int lines = 0, bars = 0, texts = 0, newViews = 0, redrawn = 0;
    Autodesk.Revit.DB.View firstView = null;

    // Colours for line styles this script makes, by bar size.
    var colours = new System.Collections.Generic.Dictionary<int, byte[]>();
    colours[8] = new byte[] { 128, 128, 128 }; colours[10] = new byte[] { 255, 127, 0 };
    colours[12] = new byte[] { 191, 127, 0 }; colours[16] = new byte[] { 0, 160, 0 };
    colours[20] = new byte[] { 0, 170, 170 }; colours[25] = new byte[] { 0, 0, 255 };
    colours[28] = new byte[] { 160, 0, 160 }; colours[32] = new byte[] { 220, 0, 0 };
    colours[40] = new byte[] { 120, 60, 0 };

    Autodesk.Revit.DB.Category linesCat =
        Autodesk.Revit.DB.Category.GetCategory(theDoc, Autodesk.Revit.DB.BuiltInCategory.OST_Lines);
    var subs = new System.Collections.Generic.Dictionary<string, Autodesk.Revit.DB.Category>();
    foreach (Autodesk.Revit.DB.Category c in linesCat.SubCategories) subs[c.Name] = c;
    var styles = new System.Collections.Generic.Dictionary<string, Autodesk.Revit.DB.GraphicsStyle>();
    var symbols = new System.Collections.Generic.Dictionary<string, Autodesk.Revit.DB.FamilySymbol>();

    // The line style for a layer key (bar-16, concrete, zones), made if missing.
    System.Func<string, Autodesk.Revit.DB.GraphicsStyle> style = delegate (string key)
    {
        Autodesk.Revit.DB.GraphicsStyle gs;
        if (styles.TryGetValue(key, out gs)) return gs;
        string name = layerName(key, "revit_line_style");
        if (string.IsNullOrWhiteSpace(name)) name = key;
        Autodesk.Revit.DB.Category sub;
        if (!subs.TryGetValue(name, out sub))
        {
            sub = theDoc.Settings.Categories.NewSubcategory(linesCat, name);
            int d = (int)num(get(layers, key), "diameter_mm");
            byte[] rgb = colours.ContainsKey(d) ? colours[d] : new byte[] { 0, 0, 0 };
            sub.LineColor = new Autodesk.Revit.DB.Color(rgb[0], rgb[1], rgb[2]);
            sub.SetLineWeight(key.StartsWith("bar-") ? 3 : 1, Autodesk.Revit.DB.GraphicsStyleType.Projection);
            subs[name] = sub;
            made.Add(name);
        }
        gs = sub.GetGraphicsStyle(Autodesk.Revit.DB.GraphicsStyleType.Projection);
        styles[key] = gs;
        return gs;
    };

    // A detail component type by "Family: Type", or by the type name alone.
    System.Func<string, Autodesk.Revit.DB.FamilySymbol> symbol = delegate (string name)
    {
        if (string.IsNullOrWhiteSpace(name)) return null;
        Autodesk.Revit.DB.FamilySymbol s;
        if (symbols.TryGetValue(name, out s)) return s;
        s = null;
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.FamilySymbol)))
        {
            var fs = e as Autodesk.Revit.DB.FamilySymbol;
            if (fs != null && (fs.FamilyName + ": " + fs.Name == name || fs.Name == name)) { s = fs; break; }
        }
        if (s == null) missing.Add(name);
        else if (!s.IsActive) s.Activate();
        symbols[name] = s;
        return s;
    };

    Autodesk.Revit.DB.ElementId textType = theDoc.GetDefaultElementTypeId(Autodesk.Revit.DB.ElementTypeGroup.TextNoteType);
    string textTypeName = layerName("text", "revit_text_type");
    if (!string.IsNullOrWhiteSpace(textTypeName))
    {
        bool found = false;
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.TextNoteType)))
            if (e.Name == textTypeName) { textType = e.Id; found = true; break; }
        if (!found) missing.Add(textTypeName);
    }

    Autodesk.Revit.DB.ElementId fillType = null;
    foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
        .OfClass(typeof(Autodesk.Revit.DB.FilledRegionType)))
    {
        if (fillType == null) fillType = e.Id;
        if (e.Name.ToUpperInvariant().Contains("SOLID")) { fillType = e.Id; break; }
    }

    // Where a point of a Triton view (mm) lands: origin + right * x + up * y.
    Autodesk.Revit.DB.XYZ origin = Autodesk.Revit.DB.XYZ.Zero;
    Autodesk.Revit.DB.XYZ right = Autodesk.Revit.DB.XYZ.BasisX;
    Autodesk.Revit.DB.XYZ up = Autodesk.Revit.DB.XYZ.BasisY;
    double dx = 0, dy = 0;
    System.Func<double, double, Autodesk.Revit.DB.XYZ> pt = delegate (double x, double y)
    { return origin + right.Multiply((x + dx) / FT) + up.Multiply((y + dy) / FT); };

    System.Action<Autodesk.Revit.DB.View, string, Autodesk.Revit.DB.XYZ, Autodesk.Revit.DB.XYZ> drawLine =
        delegate (Autodesk.Revit.DB.View view, string key, Autodesk.Revit.DB.XYZ p, Autodesk.Revit.DB.XYZ q)
        {
            if (p.DistanceTo(q) * FT < 1.0) return;   // Revit refuses shorter lines
            var line = Autodesk.Revit.DB.Line.CreateBound(p, q);
            var fam = symbol(layerName(key, "revit_line_type"));
            if (fam != null) theDoc.Create.NewFamilyInstance(line, fam, view);
            else theDoc.Create.NewDetailCurve(view, line).LineStyle = style(key);
            lines++;
        };

    System.Func<Autodesk.Revit.DB.XYZ, double, Autodesk.Revit.DB.Arc[]> halves =
        delegate (Autodesk.Revit.DB.XYZ c, double rMm)
        {
            return new Autodesk.Revit.DB.Arc[]
            {
                Autodesk.Revit.DB.Arc.Create(c, rMm / FT, 0, System.Math.PI, right, up),
                Autodesk.Revit.DB.Arc.Create(c, rMm / FT, System.Math.PI, 2 * System.Math.PI, right, up),
            };
        };

    System.Action<Autodesk.Revit.DB.View, string, Autodesk.Revit.DB.XYZ, double> drawCircle =
        delegate (Autodesk.Revit.DB.View view, string key, Autodesk.Revit.DB.XYZ c, double rMm)
        {
            foreach (var arc in halves(c, rMm)) theDoc.Create.NewDetailCurve(view, arc).LineStyle = style(key);
            lines++;
        };

    // ---- family parameters ----
    // A parameter from the file: "names" (the first of "A|B" the family has) and
    // a length "mm", a count "n" or a "text". Lengths go to Revit in feet.
    var paramNotes = new System.Collections.Generic.List<string>();
    var noted = new System.Collections.Generic.HashSet<string>();
    System.Action<string> note = delegate (string line)
    { if (noted.Add(line)) paramNotes.Add(line); };

    System.Func<Autodesk.Revit.DB.Element, object, bool> setParam = delegate (Autodesk.Revit.DB.Element el, object prm)
    {
        foreach (string n in str(prm, "names").Split('|'))
        {
            Autodesk.Revit.DB.Parameter p = el.LookupParameter(n.Trim());
            if (p == null || p.IsReadOnly) continue;
            bool hasMm = get(prm, "mm") != null, hasN = get(prm, "n") != null;
            double mm = num(prm, "mm"), cnt = num(prm, "n");
            try
            {
                if (p.StorageType == Autodesk.Revit.DB.StorageType.Double)
                {
                    bool length = true;
                    try { length = p.Definition.GetDataType() == Autodesk.Revit.DB.SpecTypeId.Length; } catch { }
                    if (hasMm) p.Set(length ? mm / FT : mm);
                    else if (hasN) p.Set(length ? cnt / FT : cnt);
                    else continue;
                }
                else if (p.StorageType == Autodesk.Revit.DB.StorageType.Integer)
                {
                    if (hasN) p.Set((int)System.Math.Round(cnt));
                    else if (hasMm) p.Set((int)System.Math.Round(mm));
                    else continue;
                }
                else if (p.StorageType == Autodesk.Revit.DB.StorageType.String)
                {
                    if (get(prm, "text") != null) p.Set(str(prm, "text"));
                    else if (hasN) p.Set(((int)cnt).ToString(inv));
                    else if (hasMm) p.Set(mm.ToString("0", inv));
                    else continue;
                }
                else continue;
                return true;
            }
            catch (System.Exception) { }
        }
        return false;
    };

    // The parameters a family has, for the summary when one is not found.
    System.Func<Autodesk.Revit.DB.Element, string> paramList = delegate (Autodesk.Revit.DB.Element el)
    {
        var names = new System.Collections.Generic.SortedSet<string>();
        foreach (Autodesk.Revit.DB.Parameter p in el.Parameters)
            if (!p.IsReadOnly) names.Add(p.Definition.Name);
        return string.Join(", ", names);
    };

    // Values on type parameters need a type of their own: one per set of
    // values, named after them, made from the family's type the first time.
    System.Func<Autodesk.Revit.DB.FamilySymbol, System.Collections.Generic.List<object>, Autodesk.Revit.DB.FamilySymbol> typeFor =
        delegate (Autodesk.Revit.DB.FamilySymbol sym, System.Collections.Generic.List<object> prms)
        {
            var onType = new System.Collections.Generic.List<object>();
            var key = new System.Collections.Generic.List<string>();
            foreach (object prm in prms)
            {
                foreach (string n in str(prm, "names").Split('|'))
                {
                    if (sym.LookupParameter(n.Trim()) == null) continue;
                    onType.Add(prm);
                    string val = get(prm, "text") != null ? str(prm, "text")
                        : get(prm, "n") != null ? ((int)num(prm, "n")).ToString(inv)
                        : num(prm, "mm").ToString("0", inv);
                    key.Add(n.Trim().Replace("DAR_", "") + " " + val);
                    break;
                }
            }
            if (onType.Count == 0) return sym;
            string name = sym.Name + " - Triton " + string.Join(" ", key);
            foreach (Autodesk.Revit.DB.ElementId id in sym.Family.GetFamilySymbolIds())
            {
                var t = theDoc.GetElement(id) as Autodesk.Revit.DB.FamilySymbol;
                if (t != null && t.Name == name) { if (!t.IsActive) t.Activate(); return t; }
            }
            var dup = sym.Duplicate(name) as Autodesk.Revit.DB.FamilySymbol;
            foreach (object prm in onType) setParam(dup, prm);
            if (!dup.IsActive) dup.Activate();
            return dup;
        };

    // Places a family item; false when it could not (the caller draws its fallback).
    System.Func<Autodesk.Revit.DB.View, object, bool> placeFamily = delegate (Autodesk.Revit.DB.View view, object it)
    {
        var sym = symbol(str(it, "family"));
        if (sym == null) return false;
        Autodesk.Revit.DB.FamilyInstance inst = null;
        try
        {
            var prms = list(it, "params");
            var typed = typeFor(sym, prms);
            Autodesk.Revit.DB.XYZ where = pt(coord(it, "at", 0), coord(it, "at", 1));
            inst = theDoc.Create.NewFamilyInstance(where, typed, view);
            foreach (object prm in prms)
            {
                if (setParam(inst, prm)) continue;
                bool onType = false;
                foreach (string n in str(prm, "names").Split('|'))
                    if (typed.LookupParameter(n.Trim()) != null) onType = true;
                if (!onType)
                    note(sym.FamilyName + ": no parameter " + str(prm, "names").Replace("|", " or ")
                        + ". It has: " + paramList(inst));
            }
            double rot = num(it, "rot");
            if (System.Math.Abs(rot) > 1e-6)
                Autodesk.Revit.DB.ElementTransformUtils.RotateElement(theDoc, inst.Id,
                    Autodesk.Revit.DB.Line.CreateBound(where, where + view.ViewDirection), rot * System.Math.PI / 180.0);
            if (str(it, "align") == "center")
            {
                theDoc.Regenerate();
                var bb = inst.get_BoundingBox(view);
                if (bb != null)
                {
                    var c = (bb.Min + bb.Max) * 0.5;
                    var move = where - c;
                    move = move - view.ViewDirection.Multiply(move.DotProduct(view.ViewDirection));
                    if (move.GetLength() > 1e-9)
                        Autodesk.Revit.DB.ElementTransformUtils.MoveElement(theDoc, inst.Id, move);
                }
            }
            return true;
        }
        catch (System.Exception ex)
        {
            note(sym.FamilyName + ": could not be placed (" + ex.Message + "), drawn with lines instead");
            if (inst != null) try { theDoc.Delete(inst.Id); } catch { }
            return false;
        }
    };

    System.Action<Autodesk.Revit.DB.View, string, Autodesk.Revit.DB.XYZ, double> drawBar =
        delegate (Autodesk.Revit.DB.View view, string key, Autodesk.Revit.DB.XYZ c, double dMm)
        {
            bars++;
            var fam = symbol(layerName(key, "revit_section_type"));
            if (fam != null)
            {
                try
                {
                    var size = new System.Collections.Generic.Dictionary<string, object>();
                    size["names"] = string.IsNullOrWhiteSpace(str(data, "bar_param")) ? "DAR_BAR DIAMETER" : str(data, "bar_param");
                    size["mm"] = dMm;
                    var prms = new System.Collections.Generic.List<object> { size };
                    var inst = theDoc.Create.NewFamilyInstance(c, typeFor(fam, prms), view);
                    if (!setParam(inst, size) && fam.LookupParameter("DAR_BAR DIAMETER") == null)
                        note(fam.FamilyName + ": no parameter " + str(size, "names").Replace("|", " or ") + ". It has: " + paramList(inst));
                    return;
                }
                catch (System.Exception ex) { note(fam.FamilyName + ": " + ex.Message + ", dots drawn instead"); }
            }
            if (fillType != null)
            {
                try
                {
                    var loop = new Autodesk.Revit.DB.CurveLoop();
                    foreach (var arc in halves(c, dMm / 2)) loop.Append(arc);
                    var region = Autodesk.Revit.DB.FilledRegion.Create(theDoc, fillType, view.Id,
                        new System.Collections.Generic.List<Autodesk.Revit.DB.CurveLoop> { loop });
                    try { region.SetLineStyleId(style(key).Id); } catch { }
                    return;
                }
                catch (Autodesk.Revit.Exceptions.ApplicationException) { }
            }
            drawCircle(view, key, c, dMm / 2);
        };

    // ---- dimensions ----
    Autodesk.Revit.DB.DimensionType dimType = null;
    string dimTypeName = layerName("dims", "revit_dimension_type");
    if (!string.IsNullOrWhiteSpace(dimTypeName))
    {
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.DimensionType)))
            if (e.Name == dimTypeName) { dimType = e as Autodesk.Revit.DB.DimensionType; break; }
        if (dimType == null) missing.Add(dimTypeName);
    }
    int dims = 0, families = 0, fallbacks = 0;

    // A Revit dimension between two short detail lines square to it; false if Revit refuses.
    System.Func<Autodesk.Revit.DB.View, object, bool> drawDim = delegate (Autodesk.Revit.DB.View view, object it)
    {
        double ax = coord(it, "a", 0), ay = coord(it, "a", 1), bx = coord(it, "b", 0), by = coord(it, "b", 1);
        double off = num(it, "off"), len = System.Math.Sqrt((bx - ax) * (bx - ax) + (by - ay) * (by - ay));
        if (len < 1.0 || System.Math.Abs(off) < 1.0) return false;
        double ux = (bx - ax) / len, uy = (by - ay) / len, nx = -uy, ny = ux;
        var witnesses = new System.Collections.Generic.List<Autodesk.Revit.DB.ElementId>();
        try
        {
            var refs = new Autodesk.Revit.DB.ReferenceArray();
            foreach (double[] q in new double[][] { new double[] { ax, ay }, new double[] { bx, by } })
            {
                var w = theDoc.Create.NewDetailCurve(view, Autodesk.Revit.DB.Line.CreateBound(
                    pt(q[0] + nx * off * 0.2, q[1] + ny * off * 0.2), pt(q[0] + nx * off, q[1] + ny * off)));
                w.LineStyle = style("dims");
                witnesses.Add(w.Id);
                refs.Append(w.GeometryCurve.Reference);
            }
            var line = Autodesk.Revit.DB.Line.CreateBound(pt(ax + nx * off, ay + ny * off), pt(bx + nx * off, by + ny * off));
            var dim = theDoc.Create.NewDimension(view, line, refs);
            if (dim == null) throw new System.InvalidOperationException("no dimension");
            if (dimType != null) dim.DimensionType = dimType;
            dims++;
            return true;
        }
        catch (System.Exception)
        {
            if (witnesses.Count > 0) try { theDoc.Delete(witnesses); } catch { }
            return false;
        }
    };

    System.Action<Autodesk.Revit.DB.View, System.Collections.Generic.List<object>> drawItems = null;
    drawItems = delegate (Autodesk.Revit.DB.View view, System.Collections.Generic.List<object> items)
    {
        foreach (object it in items)
        {
            string type = str(it, "type"), key = str(it, "layer");
            if (type == "family")
            {
                if (placeFamily(view, it)) families++;
                else { fallbacks++; drawItems(view, list(it, "fallback")); }
            }
            else if (type == "dim")
            {
                if (!drawDim(view, it)) drawItems(view, list(it, "fallback"));
            }
            else if (type == "line")
                drawLine(view, key, pt(coord(it, "a", 0), coord(it, "a", 1)), pt(coord(it, "b", 0), coord(it, "b", 1)));
            else if (type == "rect")
            {
                double ax = coord(it, "a", 0), ay = coord(it, "a", 1), bx = coord(it, "b", 0), by = coord(it, "b", 1);
                drawLine(view, key, pt(ax, ay), pt(bx, ay));
                drawLine(view, key, pt(bx, ay), pt(bx, by));
                drawLine(view, key, pt(bx, by), pt(ax, by));
                drawLine(view, key, pt(ax, by), pt(ax, ay));
            }
            else if (type == "circle")
                drawCircle(view, key, pt(coord(it, "c", 0), coord(it, "c", 1)), num(it, "r"));
            else if (type == "bar")
                drawBar(view, key, pt(coord(it, "c", 0), coord(it, "c", 1)), num(it, "d"));
            else if (type == "text")
            {
                // Triton gives the text's bottom left; a text note is placed by its top left.
                Autodesk.Revit.DB.TextNote.Create(theDoc, view.Id,
                    pt(coord(it, "at", 0), coord(it, "at", 1) + num(it, "h")), str(it, "text"), textType);
                texts++;
            }
            else if (get(it, "fallback") != null)
                drawItems(view, list(it, "fallback"));
        }
    };
    System.Action<Autodesk.Revit.DB.View, object> drawView = delegate (Autodesk.Revit.DB.View view, object v)
    { drawItems(view, list(v, "items")); };

    System.Func<string, string> clean = delegate (string name)
    {
        foreach (char bad in "\\:{}[]|;<>?`~") name = name.Replace(bad, '_');
        return name;
    };
    System.Func<Autodesk.Revit.DB.ElementId> draftingType = delegate ()
    {
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.ViewFamilyType)))
        {
            var t = e as Autodesk.Revit.DB.ViewFamilyType;
            if (t != null && t.ViewFamily == Autodesk.Revit.DB.ViewFamily.Drafting) return t.Id;
        }
        return null;
    };
    string prefix = str(data, "view_prefix");
    if (string.IsNullOrWhiteSpace(prefix)) prefix = "Triton";

    // The drafting view for a Triton view: the one of the same name, emptied, or a new one.
    System.Func<object, Autodesk.Revit.DB.View> draftingView = delegate (object v)
    {
        var parts = new System.Collections.Generic.List<string> { prefix };
        if (!string.IsNullOrWhiteSpace(str(data, "section"))) parts.Add(str(data, "section"));
        parts.Add(str(v, "name"));
        string name = clean(string.Join(" - ", parts.ToArray()));
        foreach (Autodesk.Revit.DB.Element e in new Autodesk.Revit.DB.FilteredElementCollector(theDoc)
            .OfClass(typeof(Autodesk.Revit.DB.ViewDrafting)))
        {
            var old = e as Autodesk.Revit.DB.ViewDrafting;
            if (old == null || old.IsTemplate || old.Name != name) continue;
            var ids = new System.Collections.Generic.List<Autodesk.Revit.DB.ElementId>();
            foreach (Autodesk.Revit.DB.Element x in new Autodesk.Revit.DB.FilteredElementCollector(theDoc, old.Id)
                .WhereElementIsNotElementType())
                if (x is Autodesk.Revit.DB.CurveElement || x is Autodesk.Revit.DB.TextNote || x is Autodesk.Revit.DB.Dimension
                    || x is Autodesk.Revit.DB.FamilyInstance || x is Autodesk.Revit.DB.FilledRegion)
                    ids.Add(x.Id);
            if (ids.Count > 0) theDoc.Delete(ids);
            redrawn++;
            return old;
        }
        var view = Autodesk.Revit.DB.ViewDrafting.Create(theDoc, draftingType());
        view.Name = name;
        view.Scale = System.Math.Max(1, (int)num(v, "scale"));
        newViews++;
        return view;
    };

    // Each drawing's place in one drawing of the run: its "at" (older files: a row).
    var offsets = new System.Collections.Generic.List<double[]>();
    {
        double x = 0, y = 0, rowH = 0;
        foreach (object v in chosen)
        {
            double x0 = coord(v, "box", 0), y0 = coord(v, "box", 1), x1 = coord(v, "box", 2), y1 = coord(v, "box", 3);
            if (get(v, "at") != null) { offsets.Add(new double[] { coord(v, "at", 0), coord(v, "at", 1) }); continue; }
            if (x > 0 && x + (x1 - x0) > 120000) { x = 0; y -= rowH + 6000; rowH = 0; }
            offsets.Add(new double[] { x - x0, y - y1 });
            x += (x1 - x0) + 3000;
            rowH = System.Math.Max(rowH, y1 - y0);
        }
    }
    // The top left of what was ticked, so it starts at the clicked point / the view's origin.
    double left = double.MaxValue, topY = double.MinValue;
    for (int i = 0; i < chosen.Count; i++)
    {
        left = System.Math.Min(left, coord(chosen[i], "box", 0) + offsets[i][0]);
        topY = System.Math.Max(topY, coord(chosen[i], "box", 3) + offsets[i][1]);
    }

    // Your runner may already have a transaction open; if not, this opens one.
    Autodesk.Revit.DB.Transaction tx = null;
    if (!theDoc.IsModifiable)
    {
        tx = new Autodesk.Revit.DB.Transaction(theDoc, "Triton drawings");
        tx.Start();
    }
    try
    {
        if (intoActive || intoOne)
        {
            Autodesk.Revit.DB.View target = active;
            if (intoOne)
            {
                var names = new System.Collections.Generic.SortedSet<string>();
                foreach (object v in chosen) names.Add(str(v, "element"));
                string what = names.Count == 1 ? string.Join("", names) : names.Count <= 3 ? string.Join(", ", names) : "All";
                var parts = new System.Collections.Generic.List<string> { prefix };
                if (!string.IsNullOrWhiteSpace(str(data, "section"))) parts.Add(str(data, "section"));
                parts.Add(what);
                parts.Add(System.DateTime.Now.ToString("yyyy-MM-dd HH.mm", inv));
                string name = clean(string.Join(" - ", parts.ToArray()));
                var view = Autodesk.Revit.DB.ViewDrafting.Create(theDoc, draftingType());
                for (int k = 1; k <= 20; k++)
                {
                    try { view.Name = k == 1 ? name : name + " (" + k + ")"; break; }
                    catch (Autodesk.Revit.Exceptions.ArgumentException) { }
                }
                int scale = (int)num(data, "view_scale");
                view.Scale = scale > 0 ? scale : 50;
                newViews++;
                target = view;
                firstView = view;
                origin = Autodesk.Revit.DB.XYZ.Zero;
            }
            else
            {
                origin = corner;
                right = active.RightDirection;
                up = active.UpDirection;
            }
            for (int i = 0; i < chosen.Count; i++)
            {
                dx = offsets[i][0] - left;
                dy = offsets[i][1] - topY;
                drawView(target, chosen[i]);
            }
        }
        else
        {
            foreach (object v in chosen)
            {
                var view = draftingView(v);
                if (firstView == null) firstView = view;
                dx = 0;
                dy = 0;
                drawView(view, v);
            }
        }
        if (tx != null) tx.Commit();
    }
    catch (System.Exception ex)
    {
        if (tx != null && tx.HasStarted()) tx.RollBack();
        failures.Add(ex.Message);
    }

    var done = new Autodesk.Revit.UI.TaskDialog("Triton");
    if (failures.Count > 0)
    {
        done.MainInstruction = "Nothing was drawn.";
        done.MainContent = string.Join("\n", failures.ToArray());
    }
    else
    {
        done.MainInstruction = intoActive
            ? string.Format(inv, "{0} Triton drawing(s) drawn in {1}.", chosen.Count, active.Name)
            : intoOne
                ? string.Format(inv, "{0} Triton drawing(s) drawn in the new drafting view {1}.", chosen.Count, firstView.Name)
                : string.Format(inv, "{0} drafting view(s) made, {1} drawn again.", newViews, redrawn);
        var content = new System.Text.StringBuilder();
        content.AppendLine(string.Format(inv, "{0} lines, {1} cut bars, {2} texts, {3} dimensions, {4} office families ({5} drawn with lines instead).",
            lines, bars, texts, dims, families, fallbacks));
        if (made.Count > 0)
            content.AppendLine("\nLine styles made (not in the model yet): " + string.Join(", ", made.ToArray()));
        if (missing.Count > 0)
            content.AppendLine("\nNot loaded, drawn with lines or dots instead: " + string.Join(", ", missing.ToArray()));
        if (paramNotes.Count > 0)
            content.AppendLine("\nPlease send these to Triton:\n" + string.Join("\n", paramNotes.ToArray()));
        done.MainContent = content.ToString();
    }
    done.Show();
    if (firstView != null && failures.Count == 0)
    {
        try { theUiDoc.ActiveView = firstView; } catch { }
    }
}
